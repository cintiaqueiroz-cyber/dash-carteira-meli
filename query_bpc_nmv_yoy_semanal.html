/* =====================================================================
   BPC + NMV YoY SEMANAL — Furnishing & Houseware (MLB)
   Uso: tabela semanal de Gestão de Oportunidades (MeliPro)
   ---------------------------------------------------------------------
   Convenções:
   - Semana = domingo a sábado. Rotular pelo mês do SÁBADO (fim da semana).
     Ex.: 09–15/ago => "Ago S3". Só usar semanas completas.
   - BPC (Buy Box Price Competitiveness) PIX, nível PRODUCT:
       * FAVORABILITY_TYPE = 'BOX_ALL_ABC_PIX'
       * MELI_ID_TYPE = 'PRODUCT'  (NUNCA somar ITEM/PRODUCT/UP juntos — infla o denominador)
       * BPC % = SUM(VISITS_CHEAPER + VISITS_EQUAL) / SUM(VISITS_MATCH)  (ponderado por visitas)
       * Farol: <50% vermelho | 50–65% amarelo | >65% verde
   - NMV YoY % = NMV_2026 / NMV_2025 - 1, comparando as MESMAS datas do calendário.
   - Trocar apenas as DATAS marcadas com <<< para rodar outra semana.
   ===================================================================== */

------------------------------------------------------------------
-- BLOCO 1 — BPC % da semana, por grupo (ponderado por visitas)
------------------------------------------------------------------
bpc AS (
  SELECT m.grupo,
    ROUND(SUM(f.VISITS_CHEAPER + f.VISITS_EQUAL) / NULLIF(SUM(f.VISITS_MATCH),0) * 100, 1) AS bpc_pct
  FROM meli-bi-data.WHOWNER.BT_COM_FAVORABILITY f
  JOIN mapping m ON f.CUS_CUST_ID_SEL = m.cust_id
  WHERE f.SIT_SITE_ID       = 'MLB'
    AND f.FAVORABILITY_TYPE = 'BOX_ALL_ABC_PIX'
    AND f.MELI_ID_TYPE      = 'PRODUCT'
    AND f.VERTICAL          = 'FURNISHING & HOUSEWARE'
    AND f.TIM_DAY BETWEEN DATE '2026-08-09' AND DATE '2026-08-15'   -- <<< semana (dom–sáb)
  GROUP BY m.grupo
),

------------------------------------------------------------------
-- BLOCO 2 — NMV YoY % da semana, por grupo (mesmas datas 2026 vs 2025)
------------------------------------------------------------------
nmv AS (
  SELECT m.grupo,
    SUM(IF(o.ORD_CLOSED_DT BETWEEN DATE '2026-08-09' AND DATE '2026-08-15', o.ORD_TOTAL_AMOUNT, 0)) AS nmv_2026,  -- <<< semana atual
    SUM(IF(o.ORD_CLOSED_DT BETWEEN DATE '2025-08-09' AND DATE '2025-08-15', o.ORD_TOTAL_AMOUNT, 0)) AS nmv_2025   -- <<< mesma semana 2025
  FROM meli-bi-data.WHOWNER.BT_ORD_ORDERS o
  JOIN mapping m ON o.ORD_SELLER.ID = m.cust_id
  WHERE o.SIT_SITE_ID = 'MLB'
    AND o.ORD_NMV_FLG = TRUE
    AND (o.ORD_CLOSED_DT BETWEEN DATE '2026-08-09' AND DATE '2026-08-15'
      OR o.ORD_CLOSED_DT BETWEEN DATE '2025-08-09' AND DATE '2025-08-15')
  GROUP BY m.grupo
)

------------------------------------------------------------------
-- RESULTADO — uma linha por grupo + TOTAL ponderado
------------------------------------------------------------------
SELECT
  COALESCE(b.grupo, n.grupo)                              AS grupo_de_contas,
  b.bpc_pct                                               AS bpc_semana_pct,
  ROUND((n.nmv_2026 / NULLIF(n.nmv_2025,0) - 1) * 100, 1) AS nmv_yoy_pct
FROM bpc b
FULL OUTER JOIN nmv n USING (grupo)

UNION ALL
-- Linha TOTAL FURNISHING (BPC ponderado sobre todos os cust_ids; YoY sobre a soma)
SELECT 'TOTAL FURNISHING',
  (SELECT ROUND(SUM(VISITS_CHEAPER+VISITS_EQUAL)/NULLIF(SUM(VISITS_MATCH),0)*100,1)
     FROM meli-bi-data.WHOWNER.BT_COM_FAVORABILITY f JOIN mapping m ON f.CUS_CUST_ID_SEL=m.cust_id
     WHERE f.SIT_SITE_ID='MLB' AND f.FAVORABILITY_TYPE='BOX_ALL_ABC_PIX' AND f.MELI_ID_TYPE='PRODUCT'
       AND f.VERTICAL='FURNISHING & HOUSEWARE'
       AND f.TIM_DAY BETWEEN DATE '2026-08-09' AND DATE '2026-08-15'),           -- <<< semana
  (SELECT ROUND((SUM(IF(ORD_CLOSED_DT BETWEEN DATE '2026-08-09' AND DATE '2026-08-15',ORD_TOTAL_AMOUNT,0))
               / NULLIF(SUM(IF(ORD_CLOSED_DT BETWEEN DATE '2025-08-09' AND DATE '2025-08-15',ORD_TOTAL_AMOUNT,0)),0)-1)*100,1)
     FROM meli-bi-data.WHOWNER.BT_ORD_ORDERS o JOIN mapping m ON o.ORD_SELLER.ID=m.cust_id
     WHERE o.SIT_SITE_ID='MLB' AND o.ORD_NMV_FLG=TRUE
       AND ((ORD_CLOSED_DT BETWEEN DATE '2026-08-09' AND DATE '2026-08-15')
         OR (ORD_CLOSED_DT BETWEEN DATE '2025-08-09' AND DATE '2025-08-15')))
ORDER BY grupo_de_contas;
